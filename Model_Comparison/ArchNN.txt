\begin{frame}{AI Surrogate Architectures for the Proposed Model}
\scriptsize
\textbf{Objective:} Learn the mapping between epidemic parameters and the evolution generated by the proposed mechanistic simulator.

\vspace{0.4cm}

\begin{columns}[T,onlytextwidth]
    % --- Column 1 ---
    \begin{column}{0.31\textwidth}
        \textcolor{blue!70!black}{\textbf{1. Basic Autoencoder}}
        \vspace{0.2cm}
        \begin{itemize}
            \setlength{\itemsep}{3pt}
            \item Compresses parameters into a 16-dim latent space.
            \item Minimizes MSE and peak magnitude error.
            \item Fast inference but lacks explicit temporal memory.
        \end{itemize}
    \end{column}
    
    \hfill \vrule \hfill
    
    % --- Column 2 ---
    \begin{column}{0.31\textwidth}
        \textcolor{teal!70!black}{\textbf{2. Deep AE + Smoothing}}
        \vspace{0.2cm}
        \begin{itemize}
            \setlength{\itemsep}{3pt}
            \item Deeper nonlinear layers (up to 256 units).
            \item Penalizes curve non-smoothness and peak divergence.
            \item Produces stable trajectories preserving qualitative SIR dynamics.
        \end{itemize}
    \end{column}
    
    \hfill \vrule \hfill
    
    % --- Column 3 ---
    \begin{column}{0.31\textwidth}
        \textcolor{purple!70!black}{\textbf{3. Autoregressive LSTM}}
        \vspace{0.2cm}
        \begin{itemize}
            \setlength{\itemsep}{3pt}
            \item Parameters encode the hidden state ($2 \times 128$ layers).
            \item Trajectory is generated sequentially $I(t-1) \to I(t)$.
            \item Constrains curvature with temporal derivatives.
        \end{itemize}
    \end{column}
\end{columns}

\vspace{0.6cm}
\centering
\textbf{Main Distinction:} AEs provide direct parameter-to-curve mapping, while the LSTM replicates the sequential state-to-state logic of the original simulator.
\end{frame}
